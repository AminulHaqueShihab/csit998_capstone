SELECT DATE AS gdelt_timestamp, REGEXP_EXTRACT(Extras, r'<PAGE_PRECISEPUBTIMESTAMP>(\d+)</PAGE_PRECISEPUBTIMESTAMP>') AS precise_publication_timestamp, REGEXP_EXTRACT(Extras, r'<PAGE_TITLE>(.*?)</PAGE_TITLE>') AS headline, DocumentIdentifier AS article_url, SourceCommonName AS news_source, TranslationInfo, V2Tone FROM `gdelt-bq.gdeltv2.gkg_partitioned` WHERE _PARTITIONTIME >= TIMESTAMP('2021-01-01') AND _PARTITIONTIME < TIMESTAMP('2024-01-01') AND REGEXP_CONTAINS(LOWER(REGEXP_EXTRACT(Extras, r'<PAGE_TITLE>(.*?)</PAGE_TITLE>')), r'bitcoin|btc')
